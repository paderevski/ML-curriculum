{%- comment -%} Previous / next unit links for `layout: unit` pages, taken from the "Units" children in _data/navigation.yml (the sidebar's own list). The current unit is matched by the page's source folder, so it works even if the page has a custom permalink. {%- endcomment -%} {%- assign unit_url = page.path | split: "/" | pop | join: "/" | prepend: "/" | append: "/" -%} {%- assign units = nil -%} {%- for group in site.data.navigation[page.sidebar.nav] -%} {%- for child in group.children -%} {%- if child.url == unit_url -%}{%- assign units = group.children -%}{%- endif -%} {%- endfor -%} {%- endfor -%} {%- if units -%} {%- for child in units -%} {%- if child.url == unit_url -%} {%- assign prev_i = forloop.index0 | minus: 1 -%} {%- assign next_i = forloop.index0 | plus: 1 -%} {%- endif -%} {%- endfor -%}
{%- if prev_i >= 0 -%}{%- assign u = units[prev_i] -%} Previous unit{{ u.title }} {%- endif -%} {%- if next_i < units.size -%}{%- assign u = units[next_i] -%} Next unit{{ u.title }} {%- endif -%}
{%- endif -%}
